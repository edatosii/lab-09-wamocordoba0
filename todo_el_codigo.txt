package co.umariana.conexion;

import java.io.IOException;
import java.nio.charset.StandardCharsets;
import java.nio.file.Files;
import java.nio.file.Path;
import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.SQLException;
import java.util.HashMap;
import java.util.Map;

/**
 * Conexión a MySQL configurada desde afuera del código (nunca escriba
 * contraseñas dentro del código). Cada valor se busca en este orden y gana el
 * primero que exista:
 *
 *   1. Propiedad del sistema (-D); la usan las pruebas.
 *   2. Variable de entorno.
 *   3. Archivo ".env" en la carpeta del proyecto (copie ".env.example" a ".env").
 *
 * Claves (todas obligatorias, no hay valores por defecto):
 *
 *   DB_HOST, DB_PORT, DB_NAME, DB_USER, DB_PASSWORD
 *   DB_URL   (opcional: URL JDBC completa; si existe, ignora host, puerto y nombre)
 *
 * Las PRUEBAS no usan MySQL: Maven define las propiedades db.url, db.user y
 * db.password (ver pom.xml) para apuntar a una base H2 en memoria. Por eso el
 * código de la aplicación siempre pide la conexión con ConexionBD.conectar().
 *
 * Esta clase YA está completa: no es necesario modificarla.
 */
public final class ConexionBD {

    private ConexionBD() {
    }

    private static Map<String, String> archivoEnv;

    /** Lee el archivo ".env" una sola vez. Si no existe o no se puede leer, queda vacío. */
    private static synchronized Map<String, String> archivoEnv() {
        if (archivoEnv == null) {
            archivoEnv = new HashMap<>();
            try {
                for (String linea : Files.readAllLines(Path.of(".env"), StandardCharsets.UTF_8)) {
                    linea = linea.strip(); // también quita el \r de los archivos de Windows
                    int igual = linea.indexOf('=');
                    if (linea.isEmpty() || linea.startsWith("#") || igual < 1) {
                        continue;
                    }
                    String valor = linea.substring(igual + 1).strip();
                    if (valor.length() >= 2 && (valor.startsWith("\"") && valor.endsWith("\"")
                            || valor.startsWith("'") && valor.endsWith("'"))) {
                        valor = valor.substring(1, valor.length() - 1);
                    }
                    archivoEnv.put(linea.substring(0, igual).strip(), valor);
                }
            } catch (IOException | RuntimeException e) {
                // Sin ".env" (o ilegible) se usan los valores por defecto.
            }
        }
        return archivoEnv;
    }

    /** Variable de entorno, luego archivo ".env"; null si no existe o está vacía. */
    private static String variable(String nombre) {
        String v = System.getenv(nombre);
        if (v == null || v.isBlank()) {
            v = archivoEnv().get(nombre);
        }
        return (v == null || v.isBlank()) ? null : v;
    }

    /** Propiedad del sistema (la usan las pruebas); si no existe, la variable. */
    private static String propiedadOVariable(String propiedad, String variable) {
        String v = System.getProperty(propiedad);
        return (v == null || v.isBlank()) ? variable(variable) : v;
    }

    private static String obligatoria(String propiedad, String variable) throws SQLException {
        String v = propiedadOVariable(propiedad, variable);
        if (v == null) {
            throw new SQLException("Falta la configuración " + variable
                    + ". Copie .env.example a .env (en la carpeta del proyecto) y complete todos los valores.");
        }
        return v;
    }

    private static String url() throws SQLException {
        String url = propiedadOVariable("db.url", "DB_URL");
        if (url != null) {
            return url;
        }
        return "jdbc:mysql://" + obligatoria("db.host", "DB_HOST") + ":" + obligatoria("db.port", "DB_PORT")
                + "/" + obligatoria("db.name", "DB_NAME")
                + "?useSSL=false&allowPublicKeyRetrieval=true&serverTimezone=UTC&characterEncoding=UTF-8";
    }

    /** Abre una conexión nueva. Quien la pide es responsable de cerrarla (try-with-resources). */
    public static Connection conectar() throws SQLException {
        return DriverManager.getConnection(url(),
                obligatoria("db.user", "DB_USER"),
                obligatoria("db.password", "DB_PASSWORD"));
    }
}

/**
 * DAO (Data Access Object): la única clase que habla con la base de datos.
 * Aquí viven las operaciones del CRUD sobre la tabla "estudiantes"
 * (ver db/schema.sql). Se construye igual que ClienteDAO visto en clase.
 *
 * INSTRUCCIONES PARA EL ESTUDIANTE
 * --------------------------------
 * 1. Cada método marcado con "TODO" lanza UnsupportedOperationException.
 *    Reemplace esa línea por su implementación con JDBC.
 * 2. NO modifique las pruebas (src/test). Las pruebas NO necesitan MySQL:
 *    usan una base de datos H2 en memoria.
 * 3. La Etapa 1 (registrar) YA está resuelta: úsela como modelo para las demás.
 *    Resuelva el resto en orden (Etapa 2, 3...): cada etapa usa las anteriores.
 *
 * PATRÓN (el mismo de ClienteDAO.registrar):
 *
 *     String sql = "INSERT INTO ... VALUES (?, ?, ?)";
 *     try (Connection con = ConexionBD.conectar();
 *          PreparedStatement ps = con.prepareStatement(sql)) {
 *         ps.setString(1, ...);
 *         return ps.executeUpdate() > 0;
 *     } catch (SQLException e) {
 *         System.out.println("Error al registrar: " + e.getMessage());
 *         return false;
 *     }
 *
 * REGLAS (las pruebas las verifican)
 *  - Los métodos NO lanzan SQLException: la capturan, imprimen el mensaje con
 *    System.out.println y devuelven un valor "de error" (false, lista vacía o null).
 *  - Use SIEMPRE PreparedStatement con parámetros (?), nunca concatene datos
 *    dentro del SQL (inyección SQL).
 *  - El id lo genera la base de datos (AUTO_INCREMENT): al registrar se ignora
 *    el id del objeto Estudiante.
 *  - El correo es UNIQUE: registrar o actualizar con un correo repetido provoca un
 *    SQLException; el DAO lo captura y devuelve false (no se lanza ninguna excepción).
 */
public class EstudianteDAO {

    // =====================================================================
    // ETAPA 1. Registrar (INSERT)  -- YA RESUELTA, úsela como modelo
    // =====================================================================

    /**
     * Inserta un estudiante nuevo (nombre, correo, semestre).
     *
     * @return true si se insertó; false si hubo un error (por ejemplo, correo repetido)
     */
    public boolean registrar(Estudiante estudiante) {
        String sql = "INSERT INTO estudiantes (nombre, correo, semestre) VALUES (?, ?, ?)";
        try (Connection con = ConexionBD.conectar();
             PreparedStatement ps = con.prepareStatement(sql)) {
            ps.setString(1, estudiante.getNombre());
            ps.setString(2, estudiante.getCorreo());
            ps.setInt(3, estudiante.getSemestre());
            return ps.executeUpdate() > 0;
        } catch (SQLException e) {
            System.out.println("Error al registrar: " + e.getMessage());
            return false;
        }
    }

    // =====================================================================
    // ETAPA 2. Listar (SELECT)
    // =====================================================================

/**
 * @return todos los estudiantes ordenados por id;
 *         lista vacía si no hay estudiantes o si ocurre un error.
 */
public List<Estudiante> listar() {
    List<Estudiante> estudiantes = new ArrayList<>();

    String sql = "SELECT id, nombre, correo, semestre FROM estudiantes ORDER BY id";

    try (Connection con = ConexionBD.conectar();
         PreparedStatement ps = con.prepareStatement(sql);
         ResultSet rs = ps.executeQuery()) {

        while (rs.next()) {
            Estudiante estudiante = new Estudiante(
                rs.getInt("id"),
                rs.getString("nombre"),
                rs.getString("correo"),
                rs.getInt("semestre")
            );

            estudiantes.add(estudiante);
        }

    } catch (SQLException e) {
        System.out.println("Error al listar estudiantes: " + e.getMessage());
    }

    return estudiantes;
}

    // =====================================================================
    // ETAPA 3. Buscar por id (SELECT ... WHERE)
    // =====================================================================

    /** @return el estudiante con ese id, o null si no existe (o si hay un error) */
    public Estudiante buscarPorId(int id) {
    String sql = "SELECT id, nombre, correo, semestre FROM estudiantes WHERE id = ?";

    try (Connection con = ConexionBD.conectar();
         PreparedStatement ps = con.prepareStatement(sql)) {

        ps.setInt(1, id);

        try (ResultSet rs = ps.executeQuery()) {
            if (rs.next()) {
                return new Estudiante(
                    rs.getInt("id"),
                    rs.getString("nombre"),
                    rs.getString("correo"),
                    rs.getInt("semestre")
                );
            }
        }

    } catch (SQLException e) {
        System.out.println("Error al buscar estudiante: " + e.getMessage());
    }

    return null;
}

    // =====================================================================
    // ETAPA 4. Actualizar (UPDATE)
    // =====================================================================

    /**
     * Cambia nombre, correo y semestre del estudiante cuyo id es estudiante.getId().
     *
     * @return true si se actualizó; false si no existe ese id o si hubo un error
     *         (por ejemplo, correo ya usado por OTRO estudiante)
     */
    public boolean actualizar(Estudiante estudiante) {
    String sql = "UPDATE estudiantes SET nombre = ?, correo = ?, semestre = ? WHERE id = ?";

    try (Connection con = ConexionBD.conectar();
         PreparedStatement ps = con.prepareStatement(sql)) {

        ps.setString(1, estudiante.getNombre());
        ps.setString(2, estudiante.getCorreo());
        ps.setInt(3, estudiante.getSemestre());
        ps.setInt(4, estudiante.getId());

        return ps.executeUpdate() > 0;

    } catch (SQLException e) {
        System.out.println("Error al actualizar estudiante: " + e.getMessage());
        return false;
    }
}

    // ETAPA 5. Eliminar (DELETE)
    // =====================================================================

    /** @return true si se eliminó; false si no existía ese id o si hubo un error */
    public boolean eliminar(int id) {
    String sql = "DELETE FROM estudiantes WHERE id = ?";

    try (Connection con = ConexionBD.conectar();
         PreparedStatement ps = con.prepareStatement(sql)) {

        ps.setInt(1, id);

        return ps.executeUpdate() > 0;

    } catch (SQLException e) {
        System.out.println("Error al eliminar estudiante: " + e.getMessage());
        return false;
    }

package co.umariana.servlets;

import co.umariana.dao.EstudianteDAO;
import co.umariana.modelo.Estudiante;
import java.io.IOException;
import javax.servlet.ServletException;
import javax.servlet.annotation.WebServlet;
import javax.servlet.http.HttpServlet;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpServletResponse;

/**
 * Controlador del CRUD de estudiantes: recibe las peticiones HTTP, usa el DAO
 * y decide qué vista mostrar.
 *
 * CONTRATO
 * --------------------------------------------------------------------
 * GET /estudiantes
 *   (sin accion) o accion=listar: muestra el listado de estudiantes.
 *   accion=nuevo: muestra el formulario vacío.
 *   accion=editar&id=N: busca al estudiante y muestra el formulario.
 *   Si el ID no existe o no es un número: sendError(404).
 *   Cualquier otra acción: sendError(400).
 *
 * POST /estudiantes
 *   accion=crear: registra un estudiante.
 *   accion=actualizar: actualiza un estudiante existente.
 *   accion=eliminar: elimina un estudiante.
 *   Cualquier otra acción: sendError(400).
 *
 * El nombre y el correo se guardan sin espacios al inicio/final (trim()).
 */
@WebServlet("/estudiantes")
public class EstudianteServlet extends HttpServlet {

    static final String VISTA_LISTAR = "/WEB-INF/vistas/listar.jsp";
    static final String VISTA_FORMULARIO = "/WEB-INF/vistas/formulario.jsp";

    private final EstudianteDAO dao = new EstudianteDAO();

    /**
     * Valida los datos del formulario.
     *
     * @return el mensaje de error, o null si los datos son válidos
     */
    static String validar(String nombre, String correo, String semestre) {
        if (nombre == null || nombre.trim().isEmpty()) {
            return "El nombre es obligatorio";
        }

        if (correo == null ||
            !correo.trim().matches("^[^@\\s]+@[^@\\s]+\\.[^@\\s]+$")) {
            return "El correo no es válido";
        }

        try {
            int s = Integer.parseInt(semestre.trim());

            if (s < 1 || s > 10) {
                return "El semestre debe ser un número entre 1 y 10";
            }
        } catch (NumberFormatException | NullPointerException e) {
            return "El semestre debe ser un número entre 1 y 10";
        }

        return null;
    }

    // =====================================================================
    // ETAPA 6. Mostrar (GET)
    // =====================================================================

    @Override
    protected void doGet(HttpServletRequest request,
                         HttpServletResponse response)
            throws ServletException, IOException {

        String accion = request.getParameter("accion");

        // Si no se especifica una acción o se solicita listar,
        // muestra todos los estudiantes.
        if (accion == null || "listar".equals(accion)) {
            request.setAttribute("estudiantes", dao.listar());
            request.getRequestDispatcher(VISTA_LISTAR)
                   .forward(request, response);
            return;
        }

        // Muestra el formulario vacío para registrar un estudiante nuevo.
        if ("nuevo".equals(accion)) {
            request.getRequestDispatcher(VISTA_FORMULARIO)
                   .forward(request, response);
            return;
        }

        // Busca al estudiante por su ID para mostrar el formulario de edición.
        if ("editar".equals(accion)) {
            int id;

            // Si el ID no es un número válido, responde con error 404.
            try {
                id = Integer.parseInt(request.getParameter("id"));
            } catch (NumberFormatException e) {
                response.sendError(HttpServletResponse.SC_NOT_FOUND);
                return;
            }

            // Busca al estudiante en la base de datos.
            Estudiante estudiante = dao.buscarPorId(id);

            // Si el estudiante no existe, responde con error 404.
            if (estudiante == null) {
                response.sendError(HttpServletResponse.SC_NOT_FOUND);
                return;
            }

            // Envía el estudiante a la vista para mostrar sus datos.
            request.setAttribute("estudiante", estudiante);
            request.getRequestDispatcher(VISTA_FORMULARIO)
                   .forward(request, response);
            return;
        }

        // Si la acción no está permitida, responde con error 400.
        response.sendError(HttpServletResponse.SC_BAD_REQUEST);
    }

    // =====================================================================
    // ETAPA 7. Crear, actualizar y eliminar (POST)
    // =====================================================================

    @Override
    protected void doPost(HttpServletRequest request,
                          HttpServletResponse response)
            throws ServletException, IOException {

        // Permite recibir correctamente caracteres especiales, como tildes y ñ.
        request.setCharacterEncoding("UTF-8");

        String accion = request.getParameter("accion");

        // -----------------------------------------------------------------
        // CREAR: registra un estudiante nuevo.
        // -----------------------------------------------------------------
        if ("crear".equals(accion)) {
            String nombre = request.getParameter("nombre");
            String correo = request.getParameter("correo");
            String semestre = request.getParameter("semestre");

            // Valida los datos antes de intentar guardarlos.
            String error = validar(nombre, correo, semestre);

            if (error != null) {
                request.setAttribute("error", error);
                request.getRequestDispatcher(VISTA_FORMULARIO)
                       .forward(request, response);
                return;
            }

            // Construye el objeto con los datos validados.
            Estudiante estudiante = new Estudiante(
                0,
                nombre.trim(),
                correo.trim(),
                Integer.parseInt(semestre.trim())
            );

            // Registra el estudiante y redirige al listado si tiene éxito.
            if (dao.registrar(estudiante)) {
                response.sendRedirect("estudiantes");
                return;
            }

            // Si no se pudo guardar, muestra el mensaje de error.
            request.setAttribute("error",
                "No se pudo guardar el estudiante. Verifique que el correo no esté repetido.");
            request.getRequestDispatcher(VISTA_FORMULARIO)
                   .forward(request, response);
            return;
        }

        // -----------------------------------------------------------------
        // ACTUALIZAR: modifica los datos de un estudiante existente.
        // -----------------------------------------------------------------
        if ("actualizar".equals(accion)) {
            int id;

            // Primero verifica que el ID sea un número válido.
            try {
                id = Integer.parseInt(request.getParameter("id"));
            } catch (NumberFormatException e) {
                response.sendError(HttpServletResponse.SC_NOT_FOUND);
                return;
            }

            // Comprueba que el estudiante exista antes de validar los datos.
            Estudiante existente = dao.buscarPorId(id);

            if (existente == null) {
                response.sendError(HttpServletResponse.SC_NOT_FOUND);
                return;
            }

            String nombre = request.getParameter("nombre");
            String correo = request.getParameter("correo");
            String semestre = request.getParameter("semestre");

            // Valida los nuevos datos del estudiante.
            String error = validar(nombre, correo, semestre);

            if (error != null) {
                request.setAttribute("error", error);

                // Conserva los datos enviados para volver a mostrar el formulario.
                request.setAttribute("estudiante", new Estudiante(
                    id,
                    nombre == null ? "" : nombre.trim(),
                    correo == null ? "" : correo.trim(),
                    semestreValido(semestre)
                ));

                request.getRequestDispatcher(VISTA_FORMULARIO)
                       .forward(request, response);
                return;
            }

            // Construye el objeto con el ID y los datos actualizados.
            Estudiante estudiante = new Estudiante(
                id,
                nombre.trim(),
                correo.trim(),
                Integer.parseInt(semestre.trim())
            );

            // Actualiza el estudiante y redirige al listado si tiene éxito.
            if (dao.actualizar(estudiante)) {
                response.sendRedirect("estudiantes");
                return;
            }

            // Si ocurre un error, muestra el formulario con los datos ingresados.
            request.setAttribute("error",
                "No se pudo actualizar el estudiante. Verifique que el correo no esté repetido.");
            request.setAttribute("estudiante", estudiante);
            request.getRequestDispatcher(VISTA_FORMULARIO)
                   .forward(request, response);
            return;
        }

        // -----------------------------------------------------------------
        // ELIMINAR: borra un estudiante por su ID.
        // -----------------------------------------------------------------
        if ("eliminar".equals(accion)) {
            int id;

            // Comprueba que el ID enviado sea un número válido.
            try {
                id = Integer.parseInt(request.getParameter("id"));
            } catch (NumberFormatException e) {
                response.sendError(HttpServletResponse.SC_NOT_FOUND);
                return;
            }

            // Si no se pudo eliminar, responde con error 404.
            if (!dao.eliminar(id)) {
                response.sendError(HttpServletResponse.SC_NOT_FOUND);
                return;
            }

            // Si se eliminó correctamente, vuelve al listado.
            response.sendRedirect("estudiantes");
            return;
        }

        // Cualquier acción desconocida o una petición sin acción responde 400.
        response.sendError(HttpServletResponse.SC_BAD_REQUEST);
    }

    /**
     * Convierte el semestre a un valor válido para volver a mostrar el formulario.
     *
     * @param semestre valor recibido desde el formulario
     * @return el semestre si está entre 1 y 10; en caso contrario, 1
     */
    private int semestreValido(String semestre) {
        try {
            int s = Integer.parseInt(semestre.trim());
            return (s >= 1 && s <= 10) ? s : 1;
        } catch (NumberFormatException | NullPointerException e) {
            return 1;
        }
    }
}

package co.umariana.util;

/**
 * Utilidad para las vistas JSP: escapa texto antes de imprimirlo en HTML
 * (evita ataques XSS).
 *
 * Esta clase YA está completa: no es necesario modificarla.
 */
public final class Html {

    private Html() {
    }

    public static String escapar(String texto) {
        if (texto == null) {
            return "";
        }
        return texto.replace("&", "&amp;")
                    .replace("<", "&lt;")
                    .replace(">", "&gt;")
                    .replace("\"", "&quot;")
                    .replace("'", "&#39;");
    }
}

package co.umariana.util;

import co.umariana.conexion.ConexionBD;
import java.sql.Connection;
import java.sql.SQLException;
import java.sql.Statement;

/**
 * Deja la base de datos de pruebas (H2 en memoria, con el mismo esquema que
 * MySQL: db/schema.sql) vacía y con los ids reiniciados en 1.
 * Se llama antes de cada prueba para que no se afecten entre sí.
 */
public final class BaseDeDatosDePrueba {

    private BaseDeDatosDePrueba() {
    }

    public static void limpiar() {
        try (Connection con = ConexionBD.conectar(); Statement st = con.createStatement()) {
            st.execute("DELETE FROM estudiantes");
            st.execute("ALTER TABLE estudiantes ALTER COLUMN id RESTART WITH 1");
        } catch (SQLException e) {
            throw new IllegalStateException("No se pudo limpiar la base de datos de prueba", e);
        }
    }
}
